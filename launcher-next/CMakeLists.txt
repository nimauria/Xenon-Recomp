cmake_minimum_required(VERSION 3.25)
set(XENON_NEXT_VERSION "0.1.0" CACHE STRING "Launcher package version")
project(XenonLauncherNext VERSION ${XENON_NEXT_VERSION} LANGUAGES C CXX)
if(WIN32)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|aarch64")
    set(XENON_NEXT_PLATFORM "windows-x64")
  else()
    set(XENON_NEXT_PLATFORM "windows-arm64")
  endif()
elseif(APPLE)
  set(XENON_NEXT_PLATFORM "macos-${CMAKE_SYSTEM_PROCESSOR}")
else()
  set(XENON_NEXT_PLATFORM "linux-${CMAKE_SYSTEM_PROCESSOR}")
endif()
file(WRITE "${PROJECT_BINARY_DIR}/launcher-build.json" "{\"version\":\"${PROJECT_VERSION}\",\"platform\":\"${XENON_NEXT_PLATFORM}\"}\n")
find_package(Threads REQUIRED)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(READ "${PROJECT_SOURCE_DIR}/contracts/settings.json" XENON_SETTINGS_JSON)
file(READ "${PROJECT_SOURCE_DIR}/contracts/profile-runtime.json" XENON_PROFILE_JSON)
file(READ "${PROJECT_SOURCE_DIR}/contracts/profiles.json" XENON_PROFILES_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/contracts/settings.json" "${PROJECT_SOURCE_DIR}/contracts/profile-runtime.json" "${PROJECT_SOURCE_DIR}/contracts/profiles.json")
configure_file(native/core/catalogues.hpp.in generated/catalogues.hpp @ONLY)
add_library(xenon_launcher_next_core native/core/profile_store.cpp)
target_include_directories(xenon_launcher_next_core PUBLIC native)
if(MSVC)
  target_compile_options(xenon_launcher_next_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(xenon_launcher_next_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(xenon_next_archive STATIC native/third_party/miniz/miniz.c native/third_party/miniz/miniz_tdef.c native/third_party/miniz/miniz_tinfl.c native/third_party/miniz/miniz_zip.c native/core/archive.cpp)
target_include_directories(xenon_next_archive PUBLIC native/third_party native)
add_library(xenon_next_state STATIC native/core/state_file.cpp)
target_include_directories(xenon_next_state PUBLIC native/third_party native)
add_library(xenon_launcher_next_service native/application/launcher_service.cpp native/application/capture_index.cpp native/core/game_activity.cpp native/core/game_media.cpp native/application/media_actions.cpp native/application/library_jobs.cpp native/core/file_transfer.cpp native/application/module_scan.cpp native/application/service_jobs.cpp native/adapters/host_services.cpp native/application/migration_jobs.cpp native/core/legacy_migration.cpp native/platform/legacy_storage.cpp native/platform/storage_lock.cpp native/application/install_jobs.cpp native/application/network_jobs.cpp native/application/catalog_art.cpp native/platform/desktop.cpp native/platform/http.cpp native/adapters/runtime_adapter.cpp native/core/dlc_catalog.cpp native/core/module_settings.cpp native/core/releases.cpp native/core/settings.cpp native/core/support_bundle.cpp native/core/launcher_update.cpp native/application/launcher_update_service.cpp)
target_include_directories(xenon_launcher_next_service PRIVATE "${PROJECT_BINARY_DIR}/generated")
target_compile_definitions(xenon_launcher_next_service PUBLIC XENON_LAUNCHER_VERSION="${PROJECT_VERSION}")
target_link_libraries(xenon_launcher_next_service PRIVATE xenon_next_state)
target_link_libraries(xenon_launcher_next_service PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(xenon_launcher_next_service PRIVATE shell32 winhttp advapi32 ole32 uuid)
else()
  find_package(CURL 7.85 REQUIRED)
  target_link_libraries(xenon_launcher_next_service PRIVATE CURL::libcurl)
endif()
target_link_libraries(xenon_launcher_next_service PRIVATE xenon_next_archive)
target_include_directories(xenon_launcher_next_service PUBLIC native native/third_party)
add_executable(xenon_launcher_service native/application/stdio_main.cpp)
target_link_libraries(xenon_launcher_service PRIVATE xenon_launcher_next_service)
if(WIN32)
  add_executable(xenon_launcher_updater WIN32 native/application/update_main.cpp)
else()
  add_executable(xenon_launcher_updater native/application/update_main.cpp)
endif()
target_link_libraries(xenon_launcher_updater PRIVATE xenon_launcher_next_service)
set_target_properties(xenon_launcher_updater PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/desktop")
option(XENON_NEXT_BUILD_DESKTOP "Build the separate CEF desktop launcher" OFF)
if(XENON_NEXT_BUILD_DESKTOP)
  add_subdirectory(native/host)
endif()
