cmake_minimum_required(VERSION 3.25)
project(XenonLauncherNext VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(xenon_launcher_next_core native/core/profile_store.cpp)
target_include_directories(xenon_launcher_next_core PUBLIC native)
if(MSVC)
  target_compile_options(xenon_launcher_next_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(xenon_launcher_next_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(xenon_next_archive STATIC native/third_party/miniz/miniz.c native/third_party/miniz/miniz_tdef.c native/third_party/miniz/miniz_tinfl.c native/third_party/miniz/miniz_zip.c native/core/archive.cpp)
target_include_directories(xenon_next_archive PUBLIC native/third_party native)
add_library(xenon_next_state STATIC native/core/state_file.cpp)
target_include_directories(xenon_next_state PUBLIC native/third_party native)
add_library(xenon_launcher_next_service native/application/launcher_service.cpp native/platform/desktop.cpp native/adapters/runtime_adapter.cpp)
target_link_libraries(xenon_launcher_next_service PRIVATE xenon_next_state)
if(WIN32)
  target_link_libraries(xenon_launcher_next_service PRIVATE shell32)
endif()
target_link_libraries(xenon_launcher_next_service PRIVATE xenon_next_archive)
target_include_directories(xenon_launcher_next_service PUBLIC native native/third_party)
add_executable(xenon_launcher_service native/application/stdio_main.cpp)
target_link_libraries(xenon_launcher_service PRIVATE xenon_launcher_next_service)
option(XENON_NEXT_BUILD_DESKTOP "Build the separate CEF desktop launcher" OFF)
if(XENON_NEXT_BUILD_DESKTOP)
  add_subdirectory(native/host)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(xenon_launcher_fixture_host tests/fixture_runtime_host.cpp)
  target_link_libraries(xenon_launcher_fixture_host PRIVATE xenon_next_state)
  add_executable(launcher_next_runtime_tests tests/runtime_adapter_tests.cpp)
  target_link_libraries(launcher_next_runtime_tests PRIVATE xenon_launcher_next_service)
  add_test(NAME launcher_next_runtime COMMAND launcher_next_runtime_tests $<TARGET_FILE:xenon_launcher_fixture_host>)
  set_tests_properties(launcher_next_runtime PROPERTIES TIMEOUT 30)
  add_executable(launcher_next_archive_tests tests/archive_tests.cpp)
  target_link_libraries(launcher_next_archive_tests PRIVATE xenon_next_archive)
  add_test(NAME launcher_next_archive COMMAND launcher_next_archive_tests)
  add_executable(launcher_next_service_tests tests/launcher_service_tests.cpp)
  target_link_libraries(launcher_next_service_tests PRIVATE xenon_launcher_next_service)
  add_test(NAME launcher_next_service COMMAND launcher_next_service_tests)
  add_executable(launcher_next_core_tests tests/profile_store_tests.cpp)
  target_link_libraries(launcher_next_core_tests PRIVATE xenon_launcher_next_core)
  add_test(NAME launcher_next_core COMMAND launcher_next_core_tests)
endif()
